# LiveUIN public static route semantics.
# Include these exact locations in the liveuin.com/www.liveuin.com HTTPS server
# block before the generic SPA fallback. This file is a deployment candidate;
# it does not authorize or perform a server change.

location = /sitemap.xml {
    default_type application/xml;
    try_files /sitemap.xml =404;
}

location = /robots.txt {
    default_type text/plain;
    try_files /robots.txt =404;
}

location = /liveuin-social-card.png {
    default_type image/png;
    try_files /liveuin-social-card.png =404;
}

# www.liveuin.com is a static public UI, not the Core API health origin.
# Returning a real JSON 404 prevents the SPA index from masquerading as health.
location = /api/health {
    default_type application/json;
    add_header Cache-Control "no-store" always;
    return 404 '{"ok":false,"detail":"not_found"}';
}

# Missing hashed assets must never fall back to index.html.
location ^~ /assets/ {
    try_files $uri =404;
    access_log off;
    expires 1y;
    add_header Cache-Control "public, max-age=31536000, immutable";
}
